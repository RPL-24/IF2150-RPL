Perintah: docker exec -i itbeli-db-1 psql -U itbeli -d itbeli -e < tc02.sql
Kata sandi teks asli yang dipakai akun uji: ujicoba123 (akun seed) dan Rahasia2026 (akun hasil UC01/UC02).

SELECT COUNT(*) AS total_akun,
       COUNT(*) FILTER (WHERE kata_sandi_hash ~ '^\$2[aby]\$[0-9]{2}\$[./A-Za-z0-9]{53}$') AS hash_bcrypt_valid,
       COUNT(*) FILTER (WHERE kata_sandi_hash IN ('ujicoba123', 'Rahasia2026')) AS sama_dengan_teks_asli
FROM akun;
