SELECT id_pesan, LEFT(isi_pesan_terenkripsi, 48) || '...' AS isi_tersimpan, LENGTH(isi_pesan_terenkripsi) AS panjang,
       isi_pesan_terenkripsi LIKE '%Halo kak%' AS memuat_teks_asli,
       encode(decode(isi_pesan_terenkripsi, 'base64'), 'escape') LIKE '%Halo kak%' AS teks_asli_setelah_base64_didekode
FROM pesan WHERE id_percakapan = '042e3f42-7ab0-41ff-8a4e-3c96b71b95cf';
